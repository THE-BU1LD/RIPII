"""Declared source episodes and verified manifest identity, using generated data."""
from __future__ import annotations

import hashlib
import json
import sys
from pathlib import Path

import numpy as np
import pytest
import torch

from ripii.world import cli, external_benchmark, external_data
from ripii.world.experiment import Experiment
from test_external_trajectory_data import build_dataset


GROUPED_FORMAT = "ripii-trajectory-dataset-v2"


def read_manifest(root):
    return json.loads((root / "manifest.json").read_text())


def write_manifest(root, manifest):
    (root / "manifest.json").write_text(json.dumps(manifest, sort_keys=True))


def grouped_dataset(root: Path):
    root = build_dataset(root)
    manifest = read_manifest(root)
    manifest["format"] = GROUPED_FORMAT
    manifest["source_groups"] = {}
    for split, artifact in manifest["artifacts"].items():
        with np.load(root / artifact["path"], allow_pickle=False) as archive:
            ids = archive["ids"].tolist()
        manifest["source_groups"][split] = [
            {"trajectory_id": row_id, "source_id": "source." + split, "episode_id": str(index // 2)}
            for index, row_id in enumerate(ids)
        ]
    write_manifest(root, manifest)
    return root


def test_v2_complete_dataset_verifies_and_tracks_source_groups(tmp_path):
    root = grouped_dataset(tmp_path / "dataset")
    result = external_data.verify_trajectory_dataset(root)
    assert result["format"] == GROUPED_FORMAT
    assert result["source_group_overlap_check"]["status"] == "PASS"
    for split, record in result["splits"].items():
        data, loaded = external_data.load_trajectory_split(root, split)
        assert loaded["source_groups"] == record["source_groups"]
        groups = loaded["source_groups"]
        assert groups["unique_groups"] == 2
        assert [row["trajectory_id"] for row in groups["records"]] == data["ids"].tolist()
        assert groups["records"][0]["group_sha256"] == groups["records"][1]["group_sha256"]
        assert groups["records"][0]["group_sha256"] != groups["records"][2]["group_sha256"]
        assert loaded["manifest_sha256"] == hashlib.sha256((root / "manifest.json").read_bytes()).hexdigest()


@pytest.mark.parametrize("left,right", [("train", "validation"), ("train", "test"), ("validation", "test")])
def test_shared_episode_is_rejected_before_any_archive_is_read(tmp_path, monkeypatch, left, right):
    root = grouped_dataset(tmp_path / "dataset")
    manifest = read_manifest(root)
    row = manifest["source_groups"][right][0]
    row.update({key: manifest["source_groups"][left][0][key] for key in ["source_id", "episode_id"]})
    write_manifest(root, manifest)
    monkeypatch.setattr(external_data, "_artifact_bytes", lambda *args: pytest.fail("archive opened before group admission"))
    with pytest.raises(ValueError, match="declared source episode overlaps"):
        external_data.load_trajectory_split(root, "train")


def test_additional_ood_split_participates_in_group_admission(tmp_path, monkeypatch):
    root = grouped_dataset(tmp_path / "dataset")
    manifest = read_manifest(root)
    manifest["artifacts"]["ood_shift"] = {"path": "never-open.npz"}
    first = manifest["source_groups"]["train"][0]
    manifest["source_groups"]["ood_shift"] = [{**first, "trajectory_id": -999}]
    write_manifest(root, manifest)
    monkeypatch.setattr(external_data, "_artifact_bytes", lambda *args: pytest.fail("opened OOD outcomes"))
    with pytest.raises(ValueError, match="declared source episode overlaps"):
        external_data.verify_trajectory_dataset(root)


@pytest.mark.parametrize("mutation", ["missing_split", "extra_split", "missing_row", "extra_row", "wrong_id", "duplicate_row", "cross_split_id"])
def test_exact_metadata_coverage_is_required(tmp_path, mutation):
    root = grouped_dataset(tmp_path / "dataset")
    manifest = read_manifest(root)
    groups = manifest["source_groups"]
    if mutation == "missing_split":
        del groups["validation"]
    elif mutation == "extra_split":
        groups["absent"] = groups["test"]
    elif mutation == "missing_row":
        groups["train"].pop()
    elif mutation == "extra_row":
        groups["train"].append({**groups["train"][0], "trajectory_id": -99})
    elif mutation == "wrong_id":
        groups["train"][0]["trajectory_id"] = -99
    elif mutation == "duplicate_row":
        groups["train"].append(groups["train"][0])
    else:
        groups["validation"][0]["trajectory_id"] = groups["train"][0]["trajectory_id"]
    write_manifest(root, manifest)
    with pytest.raises(ValueError, match="source_groups"):
        external_data.load_trajectory_split(root, "train")


@pytest.mark.parametrize("key,value", [("trajectory_id", True), ("trajectory_id", 1.5), ("trajectory_id", 2**63), ("source_id", ""), ("source_id", " padded"), ("source_id", "x" * 257), ("source_id", []), ("episode_id", "a\nb"), ("episode_id", None)])
def test_malformed_group_identities_are_rejected(tmp_path, key, value):

def test_group_record_follows_loaded_row_order_not_manifest_order(tmp_path):
    root = grouped_dataset(tmp_path / "dataset")
    before_data, before = external_data.load_trajectory_split(root, "train")
    manifest = read_manifest(root)
    manifest["source_groups"]["train"].reverse()
    write_manifest(root, manifest)
    after_data, after = external_data.load_trajectory_split(root, "train")
    assert before["source_groups"] == after["source_groups"]
    for key in before_data:
        assert torch.equal(before_data[key], after_data[key])


def test_v1_remains_compatible_and_does_not_infer_independence(tmp_path):
    root = grouped_dataset(tmp_path / "dataset")
    grouped, v2_record = external_data.load_trajectory_split(root, "train")
    manifest = read_manifest(root)
    manifest["format"] = external_data.FORMAT
    del manifest["source_groups"]
    write_manifest(root, manifest)
    legacy, record = external_data.load_trajectory_split(root, "train")
    assert "source_groups" not in record
    assert "source_group_overlap_check" not in external_data.verify_trajectory_dataset(root)
    for key in legacy:
        assert torch.equal(legacy[key], grouped[key])
    for key in record.keys() - {"format", "manifest_sha256"}:
        assert record[key] == v2_record[key]


def test_content_copies_are_still_rejected_with_distinct_declared_groups(tmp_path):
    root = grouped_dataset(tmp_path / "dataset")
    manifest = read_manifest(root)
    train, _ = external_data.load_trajectory_split(root, "train")
    test, _ = external_data.load_trajectory_split(root, "test")
    train["ids"] = test["ids"]
    path = root / manifest["artifacts"]["test"]["path"]
    np.savez_compressed(path, **{key: value.numpy() for key, value in train.items()})
    manifest["artifacts"]["test"].update(bytes=path.stat().st_size, sha256=hashlib.sha256(path.read_bytes()).hexdigest())
    write_manifest(root, manifest)
    with pytest.raises(ValueError, match="trajectory contents overlap"):
        external_data.verify_trajectory_dataset(root)


def test_duplicate_json_keys_cannot_override_group_declarations(tmp_path):
    root = grouped_dataset(tmp_path / "dataset")
    path = root / "manifest.json"
    text = path.read_text().replace('"source_id": "source.train"', '"source_id": "different", "source_id": "source.train"', 1)
    path.write_text(text)
    with pytest.raises(ValueError, match="duplicate JSON key"):
        external_data.load_trajectory_split(root, "train")

def test_manifest_pin_rejects_replacement_before_archive_read(tmp_path, monkeypatch):
    root = grouped_dataset(tmp_path / "dataset")
    result = external_data.verify_trajectory_dataset(root)
    manifest = read_manifest(root)
    manifest["version"] = "replaced"
    write_manifest(root, manifest)
    monkeypatch.setattr(external_data, "_artifact_bytes", lambda *args: pytest.fail("archive opened despite changed manifest"))
    with pytest.raises(ValueError, match="manifest changed since verification"):
        external_data.load_trajectory_split(root, "train", expected_manifest_sha256=result["manifest_sha256"])


@pytest.mark.parametrize("entrypoint", ["benchmark", "train_cli"])
def test_training_entrypoints_reject_manifest_changes_after_verification(tmp_path, monkeypatch, entrypoint):
    root = grouped_dataset(tmp_path / "dataset")
    module = external_benchmark if entrypoint == "benchmark" else cli
    original = external_data.verify_trajectory_dataset

    def replace_after_verification(path):
        receipt = original(path)
        manifest = read_manifest(root)
        manifest["version"] = "different-source"
        write_manifest(root, manifest)
        return receipt

    monkeypatch.setattr(module, "verify_trajectory_dataset", replace_after_verification)
    monkeypatch.setattr(module, "train", lambda *args, **kwargs: pytest.fail("training started on unverified data"))
    with pytest.raises(ValueError, match="manifest changed since verification"):
        if entrypoint == "benchmark":
            module.benchmark_trajectory_dataset(root, tmp_path / "run", Experiment(steps=1, rollout_steps=1, hidden=64), seeds=(3,), variants=("graph",))
        else:
            monkeypatch.setattr(sys, "argv", ["ripii", "train", "--dataset-root", str(root), "--output", str(tmp_path / "run"), "--steps", "1"])
            cli.main()
    assert not (tmp_path / "run").exists()
